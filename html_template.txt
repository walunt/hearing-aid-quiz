// ========== 间隔重复权重 ==========
function getWeights(){ return LS.get('weights',{}); }
function getWeight(qid){ var w=getWeights(); return (w[qid]!==undefined)?w[qid]:1.0; }
function updateWeight(qid, correct){ var w=getWeights(); var cur=(w[qid]!==undefined)?w[qid]:1.0;
  if(correct) cur=Math.max(0.15, cur*0.5); else cur=Math.min(8.0, cur*2.2); w[qid]=cur; LS.set('weights',w); }
function weightedPick(pool){ if(!pool.length) return null; var weights=[],tot=0;
  for(var i=0;i<pool.length;i++){ var ww=getWeight(pool[i]); weights.push(ww); tot+=ww; }
  var r=Math.random()*tot, cum=0; for(var i=0;i<pool.length;i++){ cum+=weights[i]; if(r<=cum) return pool[i]; } return pool[pool.length-1]; }

// ========== 渲染入口 ==========
function render(){
  document.documentElement.setAttribute('data-theme', S.theme);
  if(S.page==='home') renderHome();
  else if(S.page==='quiz') renderQuiz();
  else if(S.page==='similar') renderSimilar();
  else if(S.page==='cheat') renderCheat();
  else if(S.page==='wrong') renderWrong();
  else if(S.page==='topic') renderTopic();
}

    html += '<button class="tab-btn '+(active===tabs[i][1]?'active':'')+'" onclick="goTab(\''+tabs[i][1]+'\')">'+tabs[i][0]+'</button>';
  }
  html += '</div>';
  return html;
}

function getTopic(q){ return q.exp.title.replace(/^考察点：/,''); }

// 已做过（去重）：无论对错，答过就计入，用于算"未做题"进度

function getRounds(){ return LS.get('rounds',0); }
function addDone(qid){
  var d=getDone();
  if(d.indexOf(qid)<0){
    d.push(qid);
    if(d.length>=TOTAL){            // 本轮刷满全部题 → 完成一轮，进入下一轮并重新计数
      LS.set('rounds', getRounds()+1);
      d=[];
    }
    LS.set('done',d);
  }
}

  var rate = stats.total>0 ? (stats.correct/stats.total*100).toFixed(1) : '--';
  var doneCount = getDone().length;

function startQuiz(){ S.page='quiz'; S.history=[]; pickNext(); }
function pickNext(){
  var pool=[];
  if(S.mode==='all'){ for(var i=0;i<BANK.length;i++) pool.push(i); }
  else if(S.mode==='single'){ for(var i=0;i<BANK.length;i++) if(!BANK[i].multi) pool.push(i); }
  else if(S.mode==='multi'){ for(var i=0;i<BANK.length;i++) if(BANK[i].multi) pool.push(i); }
  else if(S.mode==='weak'){ var w=getWeights(); for(var i=0;i<BANK.length;i++){ if((w[i]||1.0)>=1.5) pool.push(i); } if(!pool.length){ for(var i=0;i<BANK.length;i++) pool.push(i); } }
  else if(S.mode==='wrong'){ pool=getWrong().slice(); if(!pool.length){ for(var i=0;i<BANK.length;i++) pool.push(i); } }
  else if(S.mode==='topic'){ if(S.topic!=null){ pool=topicPool(S.topic); if(!pool.length){ for(var i=0;i<BANK.length;i++) pool.push(i); } } }
  else if(S.mode==='undone'){ var d=getDone(); for(var i=0;i<BANK.length;i++){ if(d.indexOf(i)<0) pool.push(i); } }
  if(!pool.length){ for(var i=0;i<BANK.length;i++) pool.push(i); }
  S.qid = weightedPick(pool);
  S.answered=false; S.multi_sel={};
  render();
}
function renderQuiz(){
  var q = BANK[S.qid];
  var sessionCorrect=0, sessionTotal=S.history.length;
  for(var h=0;h<S.history.length;h++){ if(S.history[h].correct) sessionCorrect++; }
  var sessRate = sessionTotal>0 ? (sessionCorrect/sessionTotal*100).toFixed(0) : '--';
  var qWeight = getWeight(S.qid).toFixed(2);
  var last = (S.answered && S.history.length>0) ? S.history[S.history.length-1] : null;

  var mainHTML = '<div class="q-card">'+
    '<div><span class="q-tag '+(q.multi?'multi':'sel')+'">'+(q.multi?'多选题':'单选题')+'</span>'+
      '<span class="q-meta">第 '+(S.history.length+1)+' 题 · 权重 '+qWeight+'</span></div>'+
    '<div class="q-stem">'+q.s+'</div>';

  if(q.multi){
    // 多选题：可勾选，确认作答
    mainHTML += '<div class="options">';
    for(var i=0;i<q.o.length;i++){
      var k=q.o[i].k;
      var cls='multi-tog';
      if(S.answered){
        var inSel = last && last.userAns.indexOf(k)>=0;
        if(q.a.indexOf(k)>=0){ cls+=' correct'+((inSel)?'':''); }
        else if(inSel){ cls+=' wrong'; }
        else { cls+=' dim'; }
      } else {
        if(S.multi_sel[k]) cls+=' selected';
      }
      mainHTML += '<div class="option '+cls+'" '+(S.answered?'':'onclick="toggleMulti(\''+k+'\')"')+'><span class="letter">'+k+'</span><span>'+q.o[i].t+'</span></div>';
    }
    mainHTML += '</div>';
    if(!S.answered){
      mainHTML += '<div class="confirm-bar"><button class="btn btn-primary" onclick="answerMulti()">确认作答（已选 '+Object.keys(S.multi_sel).length+' 项）</button></div>';
    }
  } else {
    mainHTML += '<div class="options">';
    var labels=['A','B','C','D'];
    for(var i=0;i<q.o.length;i++){
      var cls='';
      if(S.answered){
        if(q.o[i].k===q.a) cls=' correct';
        else if(last && q.o[i].k===last.userAns) cls=' wrong';
        else cls=' dim';
      }
      mainHTML += '<div class="option'+cls+'" '+(S.answered?'':'onclick="answer(\''+q.o[i].k+'\')"')+'><span class="letter">'+q.o[i].k+'</span><span>'+q.o[i].t+'</span></div>';
    }
    mainHTML += '</div>';
  }

  if(S.answered && q.exp){
    var ok = last && last.correct;
    mainHTML += '<div class="feedback '+(ok?'ok':'err')+'">'+(ok?'✅ 回答正确！':'❌ 回答错误！')+
      '<div class="explain" style="margin-top:12px"><div class="exp-title">📖 '+q.exp.title+'</div>'+q.exp.text.replace(/\n/g,'<br>')+'</div></div>';
  }

  mainHTML += '<div class="next-hint">'+(S.answered?'点击「下一题」或按 <kbd>空格</kbd>':'点击选项作答')+'</div>'+
    (S.answered?'<div class="mobile-next"><button class="btn btn-primary" onclick="pickNext()">下一题 →</button></div>':'')+'</div>';
  var dots=''; for(var d=0;d<S.history.length;d++){ dots+='<span class="dot '+(S.history[d].correct?'ok':'bad')+'">'+(S.history[d].correct?'✓':'✗')+'</span>'; }
  var sideHTML = '<div class="side">'+
    '<div class="p-title">📊 学习面板</div>'+
    '<div class="big">'+sessRate+'%</div><div class="label">本轮正确率</div>'+
    '<div class="p-row"><span>本轮已做</span><b>'+sessionTotal+' 题</b></div>'+
    '<div class="p-row"><span>正确</span><b class="green">'+sessionCorrect+'</b></div>'+
    '<div class="p-row"><span>错误</span><b class="red">'+(sessionTotal-sessionCorrect)+'</b></div>'+
    '<div class="p-bar"><div class="p-bar-fill" style="width:'+(sessionTotal>0?sessionCorrect/sessionTotal*100:0)+'%"></div></div>'+
    '<div class="dots-wrap"><div class="p-row" style="border-bottom:none"><span>答题轨迹</span><b>'+sessionCorrect+'对 / '+(sessionTotal-sessionCorrect)+'错</b></div><div class="dots">'+(dots||'<span style="color:var(--muted);font-size:12px">暂无</span>')+'</div></div>'+
    '<div class="p-row"><span>本题权重</span><b class="'+(qWeight>=2?'red':'green')+'">'+qWeight+'</b></div>'+
    '<div class="p-title" style="margin-top:16px">🔄 本轮 / 全部进度</div>'+
    '<div class="p-row"><span>已完成轮数</span><b>'+getRounds()+' 轮</b></div>'+
    '<div class="p-row"><span>本轮进度</span><b>'+getDone().length+' / '+TOTAL+'</b></div>'+
    '<div class="p-bar"><div class="p-bar-fill" style="width:'+(TOTAL>0?getDone().length/TOTAL*100:0)+'%"></div></div>'+
    '<div class="side-actions">'+
      '<button class="btn btn-primary" onclick="pickNext()">下一题</button>'+
      '<button class="btn btn-ghost" onclick="goTab(\'home\')">结束本轮</button>'+
    '</div></div>';
  app.innerHTML = topbar('quiz') + '<div class="q-wrap">'+mainHTML+sideHTML+'</div>';
  document.documentElement.setAttribute('data-theme', S.theme);
}
function toggleMulti(k){ if(S.answered) return; if(S.multi_sel[k]) delete S.multi_sel[k]; else S.multi_sel[k]=1; renderQuiz(); }
function answer(u){
  if(S.answered) return;
  var q = BANK[S.qid];
  var correct = (u===q.a);
  S.history.push({qid:S.qid, userAns:u, correct:correct});
  S.answered=true; updateWeight(S.qid,correct);
  if(correct) removeWrong(S.qid); else addWrong(S.qid);
  addDone(S.qid);
  var stats=LS.get('stats',{total:0,correct:0}); stats.total++; if(correct) stats.correct++; LS.set('stats',stats);
  render();
}
function answerMulti(){
  if(S.answered) return;
  var q = BANK[S.qid];
  var chosenKeys=[], correct=true;
  // 逐一核对
  for(var i=0;i<q.o.length;i++){
    var k=q.o[i].k;
    var chosen = !!S.multi_sel[k];
    var right = q.a.indexOf(k)>=0;
    if(chosen!==right){ correct=false; }
    if(chosen) chosenKeys.push(k);
  }
  var ua = chosenKeys.join('');
  S.history.push({qid:S.qid, userAns:ua, correct:correct});
  S.answered=true; updateWeight(S.qid,correct);
  if(correct) removeWrong(S.qid); else addWrong(S.qid);
  addDone(S.qid);
  var stats=LS.get('stats',{total:0,correct:0}); stats.total++; if(correct) stats.correct++; LS.set('stats',stats);
  render();
}

// 完整渲染一道题（题干+全部选项+答案）
function fullQ(bq, label){
  var ansTxt;
  if(bq.multi){ ansTxt = bq.o.filter(function(o){return bq.a.indexOf(o.k)>=0}).map(function(o){return o.k+') '+o.t}).join('、'); }
  else { var o=bq.o.filter(function(x){return x.k===bq.a})[0]; ansTxt=o?o.k+') '+o.t:''; }
  var opts='';
  for(var j=0;j<bq.o.length;j++){ var o=bq.o[j]; opts+='<div class="opt'+(bq.a.indexOf(o.k)>=0?' ans':'')+'"><span class="ol">'+o.k+'</span><span>'+o.t+'</span></div>'; }
  return '<div class="fullq">'+(label?'<div class="fq-tag">'+label+'</div>':'')+
    '<div class="fq-stem">'+bq.s+'</div>'+
    '<div class="fq-opts">'+opts+'</div>'+
    '<div class="fq-ans">答案：<b>'+ansTxt+'</b></div></div>';
}
function memoSection(title, list, tag){
  var out='<div class="memo-sec"><h2>'+title+' <span class="memo-count">'+list.length+' 题</span></h2>';
  for(var i=0;i<list.length;i++){ out+=fullQ(list[i], tag); }
  return out+'</div>';
}
function renderSimilarList(q){
  var el = document.getElementById('sim-list'); if(!el) return;
  var out='';
  if(S.simFilter==='all' || S.simFilter==='memo'){
    var yl=MEMO.year.filter(function(x){return !q || x.s.indexOf(q)>=0 || x.o.some(function(o){return o.t.indexOf(q)>=0;});});
    var ll=MEMO.latin.filter(function(x){return !q || x.s.indexOf(q)>=0 || x.o.some(function(o){return o.t.indexOf(q)>=0;});});
    out += memoSection('📅 纯记忆 · 年份类', yl, '年份');
    out += memoSection('🔤 纯记忆 · 英文/数字代号', ll, '缩写');
  }
  if(S.simFilter==='all' || S.simFilter==='contrast' || S.simFilter==='variant'){
    for(var i=0;i<SIMILAR.length;i++){
      var g = SIMILAR[i];
      var isCon = g.type.indexOf('对比')>=0;
      if(S.simFilter==='contrast' && !isCon) continue;
      if(S.simFilter==='variant' && isCon) continue;
      if(q && (g.title+g.type).indexOf(q)<0 && !g.lines.some(function(it){return it[1] && it[1].indexOf(q)>=0;})) continue;
      out += '<div class="sim-group '+(isCon?'g-contrast':'g-variant')+'">'+
        '<div class="g-head"><span class="g-type">'+(isCon?'易混':'变体')+'</span><span class="g-title">'+g.title+'（'+g.type+'）</span></div>';
      var ai=0; // ans_list 游标：对比组答案行按 题A→题B 顺序
      for(var j=0;j<g.lines.length;j++){
        var it=g.lines[j];
        var isQA = it[0]==='A'||it[0]==='B';
        if(it[2]){ out += fullQ(it[2], it[0]!=='N'?it[0]:''); }
        else {
          out += '<div class="sim-item">'+(isQA?'<span class="tag">'+it[0]+'：</span>':'')+it[1]+
            (isQA&&g.ans_list&&g.ans_list[ai]?'<div class="sim-ans" style="margin-top:4px">答案：'+g.ans_list[ai]+'</div>':'')+'</div>';
        }
        if(isQA && g.ans_list) ai++;
      }
      if(g.shared_ans) out += '<div class="sim-ans">答案相同：<b>'+g.shared_ans+'</b></div>';
      if(g.variant) out += '<div class="sim-item" style="color:var(--sub)">'+g.variant+'</div>';
      if(g.diff) out += '<div class="sim-diff">⚠ '+g.diff+'</div>';
      out += '</div>';
    }
  }
  el.innerHTML = out || '<div class="card"><p>无匹配。</p></div>';
}
